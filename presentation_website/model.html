5-Line Minimal Python Inference Snippet

Python 3.8+ · pip install onnxruntime numpy 
import onnxruntime as ort
import numpy as np

# 1. Initialize quantized ONNX Runtime inference session
session = ort.InferenceSession("idr_model_submission/idr_neurospeed_tcn_int8.onnx")

# 2. Prepare 4.0-second normalized 8-channel IMU window [Batch, Channels, Timesteps]
imu_window = np.random.randn(1, 8, 40).astype(np.float32)

# 3. Predict instantaneous vehicle forward velocity (m/s) and uncertainty variance
outputs = session.run(None, {session.get_inputs()[0].name: imu_window})
velocity_mps, variance = outputs[0][0][0], outputs[0][0][1]
print(f"Estimated Vehicle Speed: {velocity_mps * 3.6:.2f} km/h (sigma: {np.sqrt(variance):.3f})")
